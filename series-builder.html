function makePreviewStore(){

  const mem={};

  const load=t=>{try{return JSON.parse(localStorage.getItem('sbpreview.'+t)||'[]')}catch(e){return mem[t]||[]}};

  const save=(t,rows)=>{mem[t]=rows;try{localStorage.setItem('sbpreview.'+t,JSON.stringify(rows))}catch(e){}};

  const rid=()=>Math.random().toString(36).slice(2)+Date.now().toString(36);

  const user={id:'preview-user',email:'preview@buildyourhearth.studio'};

  const asLead=/[?&]lead=1/.test(location.search);

  function from(t){

    const q={op:'select',vals:null,filters:[],ord:null,one:false};

    const b={

      select(){return b}, eq(c,v){q.filters.push([c,v]);return b}, single(){q.one=true;return b},

      order(c,o){q.ord=[c,!(o&&o.ascending===false)];return b},

      insert(v){q.op='insert';q.vals=v;return b}, update(v){q.op='update';q.vals=v;return b}, delete(){q.op='delete';return b},

      then(res,rej){return Promise.resolve().then(run).then(res,rej)}

    };

    const match=r=>q.filters.every(([c,v])=>r[c]===v);

    function run(){

      let rows=load(t), now=new Date().toISOString();

      if(q.op==='insert'){const r=Object.assign({id:rid(),created_at:now,updated_at:now,approved_at:now},q.vals);rows.push(r);save(t,rows);return {data:q.one?r:[r],error:null}}

      if(q.op==='update'){rows.forEach(r=>{if(match(r))Object.assign(r,q.vals,{updated_at:now})});save(t,rows);return {data:null,error:null}}

      if(q.op==='delete'){save(t,rows.filter(r=>!match(r)));return {data:null,error:null}}

      let out=rows.filter(match);

      if(q.ord){const [c,asc]=q.ord;out.sort((a,b)=>(a[c]>b[c]?1:-1)*(asc?1:-1))}

      return {data:q.one?(out[0]||null):out,error:q.one&&!out[0]?{message:'Not found'}:null};

    }

    return b;

  }

  return {

    auth:{getSession:async()=>({data:{session:{user}}}),onAuthStateChange(){},signOut:async()=>{location.reload()},signInWithOtp:async()=>({error:{message:'Preview mode'}})},

    rpc:async()=>({data:asLead,error:null}),

    from

  };

}

  const {data:{session}}=await sb.auth.getSession();

  if(session) await onSignedIn(session); else showAuth();

  sb.auth.onAuthStateChange((ev,s)=>{

    if(ev==='SIGNED_IN'&&s&&!me) onSignedIn(s);

  });

}

async function onSignedIn(session){

  me={id:session.user.id,email:session.user.email};

  $('#authView').hidden=true; $('#appView').hidden=false; setStatus('Loading…');

  const r=await sb.rpc('sb_is_lead'); isLead=!r.error&&r.data===true;

  await loadList();

  const last=localStorage.getItem('sb.last');

  const pick=list.find(x=>x.id===last)||list.find(x=>x.owner===me.id)||list[0];

  await openProject(pick.id);

}

async function loadList(){

  const {data,error}=await sb.from('sb_projects').select('id,title,owner,owner_email,updated_at').order('updated_at',{ascending:false});

  if(error){setStatus('Could not load projects');return}

  list=data||[]; renderProjects();

}

async function createProject(title,seed,open=true){

  const d=seed==='example'?{f:clone(EX.f),g:clone(EX.g),qa:{},done:{}}:{f:{},g:{},qa:{},done:{}};

  const {data,error}=await sb.from('sb_projects').insert({owner:me.id,owner_email:me.email,title:title||'Untitled series',data:d}).select('id').single();

  if(error){alert('Project not created: '+error.message);return}

  if(open){await loadList();await openProject(data.id)}

}

async function openProject(id){

  setStatus('Loading…');

  const {data,error}=await sb.from('sb_projects').select('*').eq('id',id).single();

  if(error){setStatus('Could not open project');return}

  P=data; P.data=norm(P.data); canEdit=P.owner===me.id; stage=0;

  try{localStorage.setItem('sb.last',id)}catch(e){}

  await Promise.all([loadApprovals(),loadComments()]);

  renderAll(); setStatus(canEdit?'Saved':'Viewing');

}

  approvals={}; (data||[]).forEach(a=>approvals[a.stage_id]=a);

}

}


let saveTimer=null, saving=false, again=false;

function scheduleSave(){ if(!canEdit)return; setStatus('Saving…'); clearTimeout(saveTimer); saveTimer=setTimeout(saveNow,1200) }

async function saveNow(){

  if(saving){again=true;return}

  saving=true;

  try{localStorage.setItem('sb.backup.'+P.id,JSON.stringify({title:P.title,data:P.data,at:Date.now()}))}catch(e){}

  const {error}=await sb.from('sb_projects').update({title:P.title,data:P.data}).eq('id',P.id);

  saving=false;

  setStatus(error?'Not saved. Check your connection.':'Saved');

  if(again){again=false;saveNow()}

}

function stageFields(st){const o=[];(st.cards||[]).forEach(c=>c.fields.forEach(f=>o.push(f)));return o}

function stageState(st){

  if(approvals[st.id]) return 'approved';

  if(P.data.done[st.id]) return 'done';

  const d=P.data;

  const any=stageFields(st).some(f=>filled(d.f[f.k]))||(st.groups||[]).some(g=>(d.g[g.k]||[]).some(it=>Object.values(it).some(filled)));

  return any?'prog':'empty';

}

function fieldHTML(f,val,exVal,attrs){

  const id='f_'+attrs.replace(/[^a-z0-9]/gi,'_'), ro=canEdit?'':'readonly';

  const input=f.type==='line'

    ?`<input class="line" id="${id}" ${attrs} ${ro} value="${esc(val)}">`

    :`<textarea id="${id}" ${attrs} ${ro} rows="${f.rows||3}">${esc(val)}</textarea>`;

  let meta='<span></span>';

  if(f.words){const n=wc(val);meta=`<span class="${n>f.words[1]?'warn':''}" data-wc>${n} words · target ${f.words[0]}–${f.words[1]}</span>`}

  const hasEx=filled(exVal);

  return `<div class="field"><label for="${id}">${esc(f.label)}</label>${f.hint?`<p class="hint">${esc(f.hint)}</p>`:''}${input}

function toMd(){

  const d=P.data; let md=`# ${P.title}\n\n`;

  STAGES.forEach((st,i)=>{

    md+=`## ${i+1}. ${st.title}${approvals[st.id]?' (approved)':d.done[st.id]?' (complete)':''}\n\n`;

    (st.cards||[]).forEach(c=>{md+=`### ${c.title}\n\n`;c.fields.forEach(f=>{md+=`**${f.label}**\n\n${(d.f[f.k]||'').trim()||'_Not yet written_'}\n\n`})});

    (st.groups||[]).forEach(g=>{md+=`### ${g.title}\n\n`;(d.g[g.k]||[]).forEach((it,j)=>{md+=`#### ${g.item} ${j+1}\n\n`;g.fields.forEach(f=>{md+=`**${f.label}:** ${(it[f.k]||'').trim()||'_Not yet written_'}\n\n`})})});

  });

  return md;

}

function exportMd(){

  const blob=new Blob([toMd()],{type:'text/markdown'}), a=document.createElement('a');

  a.href=URL.createObjectURL(blob); a.download=(P.title||'series').replace(/[^a-z0-9]+/gi,'-').toLowerCase()+'-bible.md';

  document.body.appendChild(a); a.click(); a.remove(); setTimeout(()=>URL.revokeObjectURL(a.href),1000);

}
